    S2 = 4'b0100, // State 2
    S3 = 4'b1000  // State 3

    case (current_state)

            O = 1'b0;
            if (A) next_state = S1;
            else next_state = S0;
        end
        S1: begin
            O = 1'b0;
            if (B) next_state = S2;
            else next_state = S1;
        end
        S2: begin
            O = 1'b0;
            if (A) next_state = S3;

        end
        S3: begin
            O = 1'b1; // Output O high in state S3
            next_state = S0; // Transition back to S0
        end
        default: next_state = S0; // Safety default
    endcase
end

// Output current state
always_ff @(posedge clk or negedge rst_n) begin
